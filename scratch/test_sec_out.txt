    function runSecurityTests() {
      var passed = 0, failed = 0;
      function test(name, cond, extra) {
        if (cond) {
          passed++;
          console.log('  [PASS] ' + name);
        } else {
          failed++;
          console.error('  [FAIL] ' + name + (extra ? ': ' + extra : ''));
        }
      }

      console.log('Starting Security, Cache Cleanup & Stale State Timeout tests...');

      try {
        var TB = window.TTBot;
        test('TTBot available', TB != null);
        test('TB.dev.cleanCaches function available', typeof TB.dev.cleanCaches === 'function');
        test('TB.dev.checkStaleStates function available', typeof TB.dev.checkStaleStates === 'function');

        var P = window.PNTT;
        var SW = P.SceneWorld;
        var g = TB.game();

        // ==========================================
        // 1. DOM XSS VULNERABILITY FIX TESTS
        // ==========================================
        console.log('--- 1. Testing DOM XSS mitigation in refreshRadar ---');
        var xssPayload = '<img src=x onerror=alert(1)>';
        var mockBoss = {
          id: 9999,
          type: 'malicious_boss',
          hp: 5000,
          hpMax: 10000,
          x: 120,
          y: 120,
          dead: false,
          def: { name: 'Quái Độc ' + xssPayload, isBoss: true }
        };
        SW.enemies = [mockBoss];

        // Evaluate refreshRadar in userscript scope
        TB.dev.eval('refreshRadar(game())');
        var radarHtml = TB.dev.eval('ui.radarInfo.innerHTML');

        test('refreshRadar escapes malicious HTML in boss name',
          radarHtml.indexOf('<img') === -1 && radarHtml.indexOf('&lt;img src=x onerror=alert(1)&gt;') >= 0,
          'radarHtml was: ' + radarHtml);

        // Test hostile player name XSS
        SW.enemies = [];
        var mockPlayer = {
          id: 'bad_actor',
          name: '<script>alert("hacked")<' + '/script>',
          x: 150,
          y: 150,
          doSat: true
        };
        P.Gateway.remotes = { bad_actor: mockPlayer };
        P.Targeting = { thuDich: function() { return true; } };
        TB.dev.eval('refreshRadar(game())');
        radarHtml = TB.dev.eval('ui.radarInfo.innerHTML');

        test('refreshRadar escapes malicious HTML in hostile player name',
          radarHtml.indexOf('<script>') === -1 && radarHtml.indexOf('&lt;script&gt;') >= 0,
          'radarHtml was: ' + radarHtml);

        // Test fallback when empty
        SW.enemies = [];
        SW.players = [];
        P.Gateway.remotes = {};
        TB.dev.eval('st.bossList = null; cfg.bossRadar = false;');
        TB.dev.eval('refreshRadar(game())');
        radarHtml = TB.dev.eval('ui.radarInfo.innerHTML');
        test('refreshRadar outputs safe fallback when empty', radarHtml === 'Không có boss / kẻ thù gần');

        // ==========================================
        // 2. PERIODIC CACHE CLEANUP TESTS
        // ==========================================
        console.log('--- 2. Testing cleanCaches for lootSkip & chatHint ---');
        var tNow = TB.dev.eval('now()');
        TB.dev.eval('lootSkip["drop_expired_1"] = ' + (tNow - 5000) + ';');
        TB.dev.eval('lootSkip["drop_expired_2"] = ' + (tNow - 100) + ';');
        TB.dev.eval('lootSkip["drop_active_1"] = ' + (tNow + 15000) + ';');

        TB.dev.eval('chatHint["than_thu_xich_long"] = { khu: 3, at: ' + (tNow - 11 * 60000) + ', dead: false };');
        TB.dev.eval('chatHint["song_duc_ma_bao"] = { khu: 1, at: ' + (tNow - 2 * 60000) + ', dead: false };');

        // Execute cache cleanup
        TB.dev.cleanCaches(true);

        var hasExp1 = TB.dev.eval('lootSkip.hasOwnProperty("drop_expired_1")');
        var hasExp2 = TB.dev.eval('lootSkip.hasOwnProperty("drop_expired_2")');
        var hasAct1 = TB.dev.eval('lootSkip.hasOwnProperty("drop_active_1")');
        test('cleanCaches purges expired lootSkip entries', !hasExp1 && !hasExp2);
        test('cleanCaches retains active lootSkip entries', hasAct1);

        var hasOldHint = TB.dev.eval('chatHint.hasOwnProperty("than_thu_xich_long")');
        var hasFreshHint = TB.dev.eval('chatHint.hasOwnProperty("song_duc_ma_bao")');
        test('cleanCaches purges chatHint older than 10 mins', !hasOldHint);
        test('cleanCaches retains fresh chatHint', hasFreshHint);

        // ==========================================
        // 3. STALE STATE TIMEOUT WATCHDOG TESTS
        // ==========================================
        console.log('--- 3. Testing checkStaleStates timeout watchdog ---');

        // Test st.bh stale timeout (exceeding 15 mins)
        TB.st.bh = {
          b: { type: 'than_thu_xich_long', spawn: 'boss_xl', map: 'thung_lung' },
          at: tNow - (16 * 60000),
          phase: 'hunt'
        };
        TB.dev.checkStaleStates(g, true);
        test('checkStaleStates auto-destructs hung st.bh (> 15 mins)', TB.st.bh == null);
        var skipXl = TB.dev.eval('bhSkip["than_thu_xich_long"]');
        test('checkStaleStates sets bhSkip after terminating hung boss hunt', skipXl > tNow);

        // Test st.bh loot phase stuck timeout (exceeding BH_LOOT_MS + 10s = 35s)
        TB.st.bh = {
          b: { type: 'song_duc_ma_bao', spawn: 'boss_mb', map: 'hoang_sa' },
          at: tNow - (2 * 60000),
          phase: 'loot',
          killAt: tNow - 40000
        };
        TB.dev.checkStaleStates(g, true);
        test('checkStaleStates auto-destructs hung st.bh stuck in loot phase (> 35s)', TB.st.bh == null);

        // Test st.bd stale timeout (exceeding 12 mins)
        TB.st.bd = {
          type: 'thach_quy_biendi',
          map: 'thung_lung',
          at: tNow - (13 * 60000)
        };
        TB.dev.checkStaleStates(g, true);
        test('checkStaleStates auto-destructs hung st.bd (> 12 mins)', TB.st.bd == null);

        // Test st.xc stale timeout (exceeding 15 mins)
        var xcEv = { map: 'thung_lung', khu: 1, what: 'Xích Nhãn Ngưu', type: 'xich_nhan_nguu', done: false };
        TB.st.xc = {
          ev: xcEv,
          map: 'thung_lung',
          at: tNow - (16 * 60000)
        };
        TB.dev.checkStaleStates(g, true);
        test('checkStaleStates auto-destructs hung st.xc (> 15 mins)', TB.st.xc == null);
        test('checkStaleStates marks event done when timing out st.xc', xcEv.done === true);

        // Test st.td stale timeout (exceeding 12 mins)
        var tdEv = { map: 'lang_tuy_tuyet', done: false };
        TB.st.td = {
          ev: tdEv,
          map: 'lang_tuy_tuyet',
          at: tNow - (13 * 60000)
        };
        TB.dev.checkStaleStates(g, true);
        test('checkStaleStates auto-destructs hung st.td (> 12 mins)', TB.st.td == null);
        test('checkStaleStates marks event done when timing out st.td', tdEv.done === true);

        // Test active non-stale state is NOT destroyed
        TB.st.bh = {
          b: { type: 'than_thu_xich_long', spawn: 'boss_xl', map: 'thung_lung' },
          at: tNow - (2 * 60000), // only 2 mins
          phase: 'hunt'
        };
        TB.dev.checkStaleStates(g, true);
        test('checkStaleStates preserves healthy active st.bh', TB.st.bh != null);
        TB.st.bh = null;

      } catch (err) {
        test('Unexpected exception during tests', false, err.stack || err.message);
      }

      console.log('----------------------------------------');
      console.log('ALL TESTS COMPLETED: ' + passed + ' PASSED, ' + failed + ' FAILED');
      document.getElementById('output').textContent = window.testLogs.join('\n') + '\n\nSUMMARY: ' + passed + ' PASSED, ' + failed + ' FAILED';
    }

    function start() {
      setTimeout(runSecurityTests, 500);
    }
    if (document.readyState === 'complete') {
      start();
    } else {
      window.addEventListener('load', start);
    }
